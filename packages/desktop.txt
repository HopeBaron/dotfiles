# The Hyprland desktop itself. Grows as components are added.
hyprland
hyprpm                      # Hyprland's own plugin manager
uwsm                        # session manager: keeps launched apps in the login
                            # session's cgroup, not user@.service. Log in via the
                            # SDDM entry "Hyprland (uwsm-managed)", not plain "Hyprland".
hyprpolkitagent             # polkit auth agent (Hyprland's own). Without one
                            # registered, pkexec falls back to a text prompt that
                            # fails with "No session for cookie" on this system --
                            # this fixes it for every app that needs polkit,
                            # not just one.
xdg-desktop-portal-hyprland
kitty
rofi                        # launcher + dmenu for window/power/clipboard menus (2.0 is Wayland-native)
jq                          # rofi-windows.sh parses `hyprctl clients -j`
waybar                      # top bar
swaync                      # notification daemon + control centre (waybar module)
swayosd                     # on-screen volume/brightness/media popup. The media keys
                            # call swayosd-client directly (keybinds.lua), so they need it
networkmanager              # waybar network module; its on-click runs nmtui
pipewire                    # audio server
wireplumber                 # pipewire session manager; wpctl (volume keys, mute toggle)
pipewire-pulse              # PulseAudio API on top of pipewire; pactl, pavucontrol, waybar
file                        # rofi-clipboard.sh detects copied MIME types with `file`
dconf                       # gsettings backend; mars-theme's theme keys need it
gsettings-desktop-schemas   # schemas mars-theme writes to (org.gnome.desktop.interface)
pavucontrol                 # waybar pulseaudio module on-click
blueman                     # waybar bluetooth module on-click
bluez                       # bluetooth stack
bluez-utils                 # bluetoothctl, used by the bluetooth module
cliphist                    # clipboard history store
wl-clipboard                # wl-copy / wl-paste, used by cliphist
imagemagick                 # thumbnails for the wallpaper/clipboard previews (lib/mars.sh)
dolphin                     # file manager, SUPER+E. Themed by kdeglobals + the
                            # CurrentTheme .colors scheme (render.sh), qt6ct QSS
yazi                        # TUI file manager, SUPER+SHIFT+E
hyprpaper                   # wallpaper daemon; misc.background_color is the fallback under it
grim                        # screenshot capture (wlr-screencopy)
slurp                       # interactive region/window picker for grim
hyprshot                    # grim+slurp wrapper; rofi-screenshot.sh drives it
breeze-icons                # fallback for icons Gruvbox-Plus-Dark (aur.txt) lacks
qt6ct                       # Qt6 theming; QT_QPA_PLATFORMTHEME=qt6ct in
                            # hyprland.lua, scheme generated by render.sh
hyprlock                    # lock screen; the power menu's Lock entry
hypridle                    # idle daemon: dims/locks/suspends, autostarted from hyprland.lua
sddm                        # login screen, replaces EndeavourOS's GDM. install.sh copies
                            # system/sddm/ into /usr/share and /etc and switches the DM.
